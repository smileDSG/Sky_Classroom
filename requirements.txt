# Node.js (必需)
#   推荐版本: 22.x (LTS)  —  最低版本: >= 20.9.0
#   下载: https://nodejs.org/
#   版本管理: 可使用 nvm / fnm / volta，项目根目录 .nvmrc 指定 v22
NODE_VERSION>=20.9.0

# pnpm (必需)
#   推荐版本: 10.x  —  项目锁定: 10.28.0
#   安装: corepack enable && corepack prepare pnpm@10.28.0 --activate
#   或: npm install -g pnpm@10
PNPM_VERSION>=10.0.0

# 2. 系统级依赖 (System Dependencies)
# -----------------------------------------------------------------------------

# --- 构建时依赖 (用于 native modules: sharp, @napi-rs/canvas, better-sqlite3) ---
# Debian / Ubuntu:
#   apt-get install -y python3 build-essential g++ \
#     libcairo2-dev libpango1.0-dev libjpeg-dev libgif-dev librsvg2-dev
#
# Alpine (Docker dev):
#   apk add --no-cache libc6-compat python3 build-base g++ \
#     cairo-dev pango-dev jpeg-dev giflib-dev librsvg-dev
#
# macOS:
#   brew install python3 pkg-config cairo pango jpeg giflib librsvg
#
# Windows:
#   npm install -g windows-build-tools
#   或安装 Visual Studio Build Tools (C++ 桌面开发工作负载)

# --- 运行时依赖 (生产环境 / PPTX 转图片) ---
# LibreOffice (可选，PPTX→PNG 导出需要):
#   Debian/Ubuntu: apt-get install -y libreoffice
#   macOS: brew install --cask libreoffice
#   Windows: 下载安装 https://www.libreoffice.org/
#
# 运行时 native 库 (与构建时对应的 .so/.dll):
#   libcairo2 libpango-1.0-0 libjpeg62-turbo libgif7 librsvg2-2

# -----------------------------------------------------------------------------
# 7. 快速启动命令
# -----------------------------------------------------------------------------

# git clone https://github.com/THU-MAIC/OpenMAIC.git
# cd OpenMAIC
# pnpm install
# cp .env.example .env.local      # 编辑填入 API Key
# pnpm dev                         # 启动开发服务器 → http://localhost:3000

# 生产构建:
# pnpm build && pnpm start

# Docker 部署:
# docker compose up --build

# 运行测试:
# pnpm test                        # 单元测试
# pnpm test:e2e                    # E2E 测试
# pnpm lint                        # ESLint 检查
# pnpm format                      # Prettier 格式化
